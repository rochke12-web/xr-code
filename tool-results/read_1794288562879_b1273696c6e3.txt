   138→   138→  try {

   140→   140→    if (
   141→   141→      (o.action === "image" || o.action === "video") &&
   142→   142→      typeof o.prompt === "string" && o.prompt.trim().length > 2

   241→   241→function wantsWeb(q: string): boolean {
   242→   242→  const s = q.toLowerCase();
   243→   243→  if (s.length < 4) return false;
   244→   244→  // identity / meta questions about the AI itself — answer directly, never search
   245→   245→  if (/(who are you|what are you|your name|who made you|who created you|who built you|about yourself|who are u)/.test(s)) return false;
   246→   246→  const questionWord = /^(who|what|when|where|why|how|which|whose|is|are|was|were|do|does|did|can|could|should|define|explain|tell me about)\b/.test(s);
   247→   247→  const currentInfo = /(latest|current|today|right now|news|price of|net worth|population|how old|release date|who is|what is)/.test(s);
   248→   248→  const followUp = /^(and|also|his|her|its|their|what about|how about)\b/.test(s) && s.includes("?");
   249→   249→  if (!(questionWord || currentInfo || followUp)) return false;
   250→   250→  // pure code tasks never search, even if phrased oddly long
   251→   251→  if (/^(write|create|make|build|fix|refactor|optimi[sz]e|implement|add)\b/.test(s) && !s.includes("?") && !/who is|what is|latest|current|news/.test(s)) return false;
   252→   252→  return true;
   253→   253→}

   348→   348→  );
   349→   349→  const heavy = s.length >= 140 || tech.test(s) || deepTech;
   350→   350→  return heavy ? GLM_FLAGSHIP : GLM_QUICK;
   351→   351→}

   353→   353→// free GLM engines via the ZAI SDK (OpenAI-style messages, verified live)
   354→   354→async function glmChat(sys: string, msgs: { role: "user" | "assistant"; content: string }[], model: string): Promise<string | null> {
   355→   355→  // self-healing: up to 2 attempts, re-creating the engine client each time —
   356→   356→  // a stale/poisoned SDK instance (env refresh after recycle) recovers instantly.
   357→   357→  // (was 3 — retries amplify gateway 429 storms, making outages last longer)
   358→   358→  for (let attempt = 0; attempt < 2; attempt++) {
   359→   359→    try {
   360→   360→      const zai = await ZAI.create();
   361→   361→      const res = (await Promise.race([
   362→   362→        zai.chat.completions.create({
   363→   363→          model,
   364→   364→          messages: [{ role: "system", content: sys }, ...msgs],
   365→   365→          thinking: { type: "disabled" as const },
   366→   366→        }),
   367→   367→        new Promise<null>(r => setTimeout(() => r(null), 60000)),
   368→   368→      ])) as { choices?: { message?: { content?: string } }[] } | null;
   369→   369→      const text = String(res?.choices?.[0]?.message?.content || "").trim();
   370→   370→      if (text) return text;
   371→   371→    } catch {
   372→   372→      // engine hiccup — fall through to retry
   373→   373→    }
   374→   374→    if (attempt < 1) await new Promise(r => setTimeout(r, 1200));
   375→   375→  }
   376→   376→  return null; // caller tries the next engine
   377→   377→}

   379→   379→// pull plain text out of string-or-multipart content
   380→   380→function textOf(content: unknown): string {
   381→   381→  if (typeof content === "string") return content;
   382→   382→  if (Array.isArray(content)) {
   383→   383→    return content
   384→   384→      .map(p => (p && typeof p === "object" && typeof (p as Part).text === "string" ? ((p as Part).text as string) : ""))
   385→   385→      .join("\n");
   386→   386→  }
   387→   387→  return "";
   388→   388→}

   390→   390→// ---- real vision preprocessing ----
   391→   391→// pull base64 image blocks (photos + video keyframes) out of the last user message
   392→   392→function imagesOf(content: unknown): { mt: string; b64: string }[] {
   393→   393→  if (!Array.isArray(content)) return [];
   394→   394→  const out: { mt: string; b64: string }[] = [];
   395→   395→  for (const p of content) {
   396→   396→    const part = p as Part;
   397→   397→    const src = part?.source;
   398→   398→    if (part?.type === "image" && src && typeof src.data === "string" && src.data.length > 64) {
   399→   399→      const mt =
   400→   400→        src.media_type && /^image\/(png|jpe?g|webp|gif)$/.test(src.media_type)
   401→   401→          ? (src.media_type === "image/jpg" ? "image/jpeg" : src.media_type)
   402→   402→          : "image/jpeg";
   403→   403→      if (src.data.length <= 4_200_000) out.push({ mt, b64: src.data });
   404→   404→    }
   405→   405→  }
   406→   406→  return out.slice(0, 8);
   407→   407→}

   409→   409→// describe the images with a vision model; returns one 'Image N: …' line each
   410→   410→async function describeImages(imgs: { mt: string; b64: string }[]): Promise<string> {
   411→   411→  if (!imgs.length) return "";
   412→   412→  try {
   413→   413→    const zai = await ZAI.create();
   414→   414→    const content: Extract<VisionMessage["content"], unknown[]> = [
   415→   415→      {
   416→   416→        type: "text",
   417→   417→        text:
   418→   418→          "You are the vision preprocessor of a chat app. For EACH attached image, in order, " +
   419→   419→          "write exactly one line: 'Image N: <description>'. Describe concretely what is shown " +
   420→   420→          "(objects, colors, scene, layout) and transcribe ALL visible text verbatim. " +
   421→   421→          "No preamble, no extra lines.",
   422→   422→      },
   423→   423→    ];
   424→   424→    for (const im of imgs) {
   425→   425→      content.push({ type: "image_url", image_url: { url: `data:${im.mt};base64,${im.b64}` } });
   426→   426→    }
   427→   427→    // SDK types demand `model` and omit `role`, but the verified-working runtime
   428→   428→    // payload needs neither — cast instead of changing what works.
   429→   429→    const visionBody = {
   430→   430→      messages: [{ role: "user", content }],
   431→   431→      thinking: { type: "disabled" as const },
   432→   432→    };
   433→   433→    const res = (await Promise.race([
   434→   434→      zai.chat.completions.createVision(
   435→   435→        visionBody as unknown as Parameters<typeof zai.chat.completions.createVision>[0]
   436→   436→      ),
   437→   437→      new Promise<null>(r => setTimeout(() => r(null), 32000)),
   438→   438→    ])) as { choices?: { message?: { content?: string } }[] } | null;
   439→   439→    return String(res?.choices?.[0]?.message?.content || "").trim();
   440→   440→  } catch {
   441→   441→    return ""; // vision preprocessing down → fall back to blind text notes
   442→   442→  }
   443→   443→}

   449→   449→export async function POST(req: Request) {
   450→   450→  let body: { messages?: RawMsg[]; model?: string; mode?: string } | null = null;
   451→   451→  try {
   452→   452→    body = await req.json();
   453→   453→  } catch {
   454→   454→    /* no / invalid body */
   455→   455→  }
   456→   456→  const raw = Array.isArray(body?.messages) ? (body!.messages as RawMsg[]) : null;
   457→   457→  const safe: Msg[] = raw
   458→   458→    ? raw
   459→   459→        .slice(-24)
   460→   460→        .filter(m => m && (m.role === "user" || m.role === "assistant") && m.content != null)
   461→   461→        .map(m => ({ role: m.role as "user" | "assistant", content: textOf(m.content).slice(0, 8000) }))
   462→   462→    : [];

   464→   464→    return NextResponse.json({ ok: false, error: "bad request" }, { status: 400 });
   465→   465→  }
   466→   466→
   467→   467→  // two modes: Koder (developer brain) and QnA (web-grounded answers, sources stay hidden)
   468→   468→  const modeRaw = typeof body?.mode === "string" ? body.mode.trim().toLowerCase() : "";
   469→   469→  const mode = modeRaw === "qna" || modeRaw === "sight" ? modeRaw : "koder";
   470→   470→  let sys = (mode === "qna" ? SYSTEM_QNA : mode === "sight" ? SYSTEM_SIGHT : SYSTEM_KODER) + TOOL_RULES;
   471→   471→  const lastUser = [...safe].reverse().find(m => m.role === "user");
   472→   472→  // one brain, all GLM: explicit family member is honored, otherwise auto-route

   474→   474→  const chosen = GLM_MODELS.has(wanted) ? wanted : autoEngine(lastUser?.content || "", safe);
   475→   475→  // the user is talking about THEIR attachment — web search would only distract.
