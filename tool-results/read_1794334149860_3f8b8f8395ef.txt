   163→// a real directive is a SHORT reply that IS a JSON object — long answers with a
   164→// JSON snippet inside are just code, never a tool call
   165→function parseAction(reply: string | null): Action | null {
   166→  if (!reply) return null;
   167→  const s = reply.trim().replace(/^```[a-z]*\s*/i, "").replace(/\s*```$/, "").trim();
   168→  if (s.length > 600 || !s.startsWith("{")) return null;
   169→  try {
   170→    const o = JSON.parse(s) as Record<string, unknown>;
   171→    if (o.action === "search" && typeof o.query === "string" && o.query.trim().length > 1) {
   172→      return { action: "search", query: o.query.trim().slice(0, 200) };
   173→    }
   174→    if (
   175→      (o.action === "image" || o.action === "video") &&
   176→      typeof o.prompt === "string" && o.prompt.trim().length > 2
   177→    ) {
   178→      return {
   179→        action: o.action,
   180→        prompt: o.prompt.trim().slice(0, 800),
   181→        size: typeof o.size === "string" ? o.size : undefined,
   182→        with_audio: o.with_audio === true,
   183→      };
   184→    }
   185→  } catch {
   186→    /* normal text */
   187→  }
   188→  return null;
   189→}

   273→// does this message deserve live web results?
   274→// knowledge / current-info questions — yes. pure code tasks and math/reasoning — no.
   275→function wantsWeb(q: string): boolean {
   276→  const s = q.toLowerCase();
   277→  if (s.length < 4) return false;
   278→  // identity / meta questions about the AI itself — answer directly, never search
   279→  if (/(who are you|what are you|your name|who made you|who created you|who built you|about yourself|who are u)/.test(s)) return false;
   280→  const questionWord = /^(who|what|when|where|why|how|which|whose|is|are|was|were|do|does|did|can|could|should|define|explain|tell me about)\b/.test(s);
   281→  const currentInfo = /(latest|current|today|right now|news|price of|net worth|population|how old|release date|who is|what is)/.test(s);
   282→  const followUp = /^(and|also|his|her|its|their|what about|how about)\b/.test(s) && s.includes("?");
   283→  if (!(questionWord || currentInfo || followUp)) return false;
   284→  // pure code tasks never search, even if phrased oddly long
   285→  if (/^(write|create|make|build|fix|refactor|optimi[sz]e|implement|add)\b/.test(s) && !s.includes("?") && !/who is|what is|latest|current|news/.test(s)) return false;
   286→  return true;
   287→}

   289→// context-aware search query: short follow-ups like "how old is he?" inherit
   290→// the previous question's topic so the search still finds the right thing.
   291→function searchQuery(safe: Msg[]): string {
   292→  const users = safe.filter(m => m.role === "user");
   293→  const last = (users[users.length - 1]?.content ?? "").trim();
   294→  if (
   295→    users.length >= 2 &&
   296→    (last.length < 45 || /^(and|also|his|her|its|their|what about|how about|then|so)\b/i.test(last))
   297→  ) {
   298→    const prev = users[users.length - 2].content.slice(0, 120);
   299→    return `${prev} ${last}`.slice(0, 300);
   300→  }
   301→  return last.slice(0, 300);
   302→}

   311→const SEARCH_UA =

   314→function decodeEntities(s: string): string {
   315→  return s
   316→    .replace(/&amp;/g, "&").replace(/&lt;/g, "<").replace(/&gt;/g, ">")
   317→    .replace(/&quot;/g, '"').replace(/&#0?39;|&apos;|&#x27;/g, "'")
   318→    .replace(/&nbsp;/g, " ").replace(/&#(\d+);/g, (_, d) => String.fromCharCode(Number(d)));
   319→}
   320→function stripTags(s: string): string {
   321→  return decodeEntities(String(s).replace(/<[^>]*>/g, "")).replace(/\s+/g, " ").trim();
   322→}

   324→async function fetchPage(url: string, ms: number): Promise<string | null> {
   325→  try {
   326→    const ctrl = new AbortController();
   327→    const timer = setTimeout(() => ctrl.abort(), ms);
   328→    const res = await fetch(url, {
   329→      signal: ctrl.signal,
   330→      redirect: "follow",
   331→      headers: { "User-Agent": SEARCH_UA, "Accept-Language": "en-US,en;q=0.9" },
   332→    });
   333→    clearTimeout(timer);
   334→    if (!res.ok) return null;
   335→    return await res.text();
   336→  } catch {
   337→    return null;
   338→  }
   339→}

   341→// Google News RSS — real Google results, no JS wall, no API key
   342→async function googleNews(q: string): Promise<SearchItem[]> {
   343→  const xml = await fetchPage(
   344→    `https://news.google.com/rss/search?q=${encodeURIComponent(q)}&hl=en-US&gl=US&ceid=US:en`, 9000);
   345→  if (!xml) return [];
   346→  const out: SearchItem[] = [];
   347→  for (const chunk of xml.split("<item>").slice(1, 9)) {
   348→    const name = stripTags(chunk.match(/<title>([\s\S]*?)<\/title>/)?.[1] || "");
   349→    const url = (chunk.match(/<link>([\s\S]*?)<\/link>/)?.[1] || "").trim();
   350→    const date = (chunk.match(/<pubDate>([\s\S]*?)<\/pubDate>/)?.[1] || "").trim();
   351→    if (!name || !url) continue;
   352→    out.push({ name, snippet: name, url, date: date ? date.slice(5, 16) : undefined });
   353→  }
   354→  return out;
   355→}

   357→// DuckDuckGo HTML — general-web gap filler (Google News skews to current events)
   358→async function duckWeb(q: string): Promise<SearchItem[]> {
   359→  let html: string | null = null;
   360→  for (let i = 0; i < 2 && !html; i++) {
   361→    html = await fetchPage(`https://html.duckduckgo.com/html/?q=${encodeURIComponent(q)}`, 8000);
   362→  }
   363→  if (!html) return [];
   364→  const out: SearchItem[] = [];
   365→  const re = /<a[^>]*class="[^"]*result__a[^"]*"[^>]*href="([^"]+)"[^>]*>([\s\S]*?)<\/a>/g;
   366→  const snips = [...html.matchAll(/<a[^>]*class="[^"]*result__snippet[^"]*"[^>]*>([\s\S]*?)<\/a>/g)]
   367→    .map(m2 => stripTags(m2[1]));
   368→  let m: RegExpExecArray | null;
   369→  while ((m = re.exec(html)) && out.length < 8) {
   370→    let url = m[1];
   371→    const uddg = url.match(/[?&]uddg=([^&]+)/);
   372→    if (uddg) { try { url = decodeURIComponent(uddg[1]); } catch { /* keep raw */ } }
   373→    const name = stripTags(m[2]);
   374→    if (!name || !/^https?:\/\//.test(url) || /duckduckgo\.com/.test(url)) continue;
   375→    out.push({ name, snippet: snips[out.length] || name, url });
   376→  }
   377→  return out;
   378→}

   402→// merge: Google News first, DuckDuckGo web second, hosted index as the safety net
   403→async function googleSearch(q: string): Promise<SearchItem[]> {
   404→  const [news, web] = await Promise.all([googleNews(q), duckWeb(q)]);
   405→  const seen = new Set<string>();
   406→  const merged: SearchItem[] = [];
   407→  for (const it of [...news, ...web]) {
   408→    const key = it.url.replace(/[?#].*$/, "").toLowerCase();
   409→    if (!it.url || seen.has(key)) continue;
   410→    seen.add(key);
   411→    merged.push(it);
   412→    if (merged.length >= 8) break;
   413→  }
   414→  if (merged.length < 3) {
   415→    for (const it of await hostedSearch(q)) {
   416→      const key = it.url.replace(/[?#].*$/, "").toLowerCase();
   417→      if (seen.has(key)) continue;
   418→      seen.add(key);
   419→      merged.push(it);
   420→      if (merged.length >= 8) break;
   421→    }
   422→  }
   423→  return merged;
   424→}

   427→async function googleContext(q: string): Promise<{ block: string; items: SearchItem[] }> {
   428→  const items = await googleSearch(q);

   431→    (it, i) =>
   432→      `${i + 1}. ${it.name}${it.date ? ` (${it.date})` : ""}\n` +

   434→      `   source: ${it.url}`

   437→    "\n\nLIVE WEB RESULTS (Google search) for this question:\n" +
   438→    lines.join("\n") +

   440→    "top result would read — same facts, but with light grammar and wording changes so it is " +
   441→    "not a verbatim copy. If the results do not actually answer the question, answer from your " +
   442→    "own knowledge instead. Be direct and concise, no markdown headings. " +
   443→    "Do not write a Sources line.";

   500→async function glmChat(sys: string, msgs: { role: "user" | "assistant"; content: string }[], model: string): Promise<ChatResult> {
   501→  // self-healing: up to 2 attempts, re-creating the engine client each time —
   502→  // a stale/poisoned SDK instance (env refresh after recycle) recovers instantly.
   503→  // 429 short-circuit: return quota:true at once — the CALLER decides whether
   504→  // to probe the other free lanes (FALLBACK_MODELS) or answer from Google /
   505→  // honest outage. No unbounded retries here: one message must never spin
   506→  // 16+ gateway calls.
   507→  for (let attempt = 0; attempt < 2; attempt++) {
   508→    try {
   509→      const zai = await ZAI.create();
   510→      // thinking toggle is a GLM knob — applied to glm-* models only, never
   511→      // to the emergency dpsk/argon lanes
   512→      const glmish = model.startsWith("glm");
   513→      const res = (await Promise.race([
   514→        zai.chat.completions.create({
   515→          model,
   516→          messages: [{ role: "system", content: sys }, ...msgs],
   517→          ...(glmish ? { thinking: { type: "disabled" as const } } : {}),
   518→        }),
   519→        new Promise<null>(r => setTimeout(() => r(null), 60000)),
   520→      ])) as { choices?: { message?: { content?: string } }[] } | null;
   521→      const text = String(res?.choices?.[0]?.message?.content || "").trim();
   522→      if (text) return { text, quota: false };
   523→      // empty reply is an engine hiccup, not a quota signal — retry once
   524→    } catch (e) {
   525→      const msg = String((e as Error)?.message || e);
   526→      if (/\b429\b|too many requests/i.test(msg)) return { text: null, quota: true };
   527→      // engine hiccup — fall through to retry
   528→    }
   529→    if (attempt < 1) await new Promise(r => setTimeout(r, 1200));
   530→  }
   531→  return { text: null, quota: false }; // caller tries the next engine
   532→}

   549→// pull plain text out of string-or-multipart content
   550→function textOf(content: unknown): string {
   551→  if (typeof content === "string") return content;
   552→  if (Array.isArray(content)) {
   553→    return content
   554→      .map(p => (p && typeof p === "object" && typeof (p as Part).text === "string" ? ((p as Part).text as string) : ""))
   555→      .join("\n");
   556→  }
   557→  return "";

   560→// ---- real vision preprocessing ----
   561→// pull base64 image blocks (photos + video keyframes) out of the last user message
   562→function imagesOf(content: unknown): { mt: string; b64: string }[] {
   563→  if (!Array.isArray(content)) return [];
   564→  const out: { mt: string; b64: string }[] = [];
   565→  for (const p of content) {
   566→    const part = p as Part;
   567→    const src = part?.source;
   568→    if (part?.type === "image" && src && typeof src.data === "string" && src.data.length > 64) {
   569→      const mt =
   570→        src.media_type && /^image\/(png|jpe?g|webp|gif)$/.test(src.media_type)
   571→          ? (src.media_type === "image/jpg" ? "image/jpeg" : src.media_type)
   572→          : "image/jpeg";
   573→      if (src.data.length <= 4_200_000) out.push({ mt, b64: src.data });
   574→    }
   575→  }
   576→  return out.slice(0, 8);
   577→}

   821→  for (const engine of chain) {
   822→    const r = await glmChat(sys, chatText, engine);
   823→    if (r.quota) {
   824→      if (engine !== chosen) continue; // fallback lane dry — next lane
   825→      // primary is throttled: one grace retry, then the remaining lanes
   826→      await new Promise(r2 => setTimeout(r2, 3000));
   827→      const retry = await glmChat(sys, chatText, chosen);
   828→      if (retry.text) {
   829→        const a2 = parseAction(retry.text);
   830→        if (a2) { actText = retry.text; engineUsed = chosen; act = a2; break; }
   831→        return NextResponse.json({ ok: true, text: withSources(retry.text, mode, lastSearch), engine: "xr-code", mode, vision: visionBlock || undefined });
   832→      }
   833→      continue;
   834→    }
   835→    if (!r.text) continue;
   836→    const t = r.text;
   837→    const a = parseAction(t);
   838→    if (a) {
   839→      actText = t;
   840→      engineUsed = engine;
   841→      act = a;
   842→      break;
   843→    }
   844→    return NextResponse.json({ ok: true, text: withSources(t, mode, lastSearch), engine: "xr-code", mode, vision: visionBlock || undefined });
   845→  }
   846→  if (!actText || !act) {
   847→    const out = googleOutage();
   848→    if (out) return out;
   849→    return NextResponse.json({ ok: true, text: outageReply(mode), engine: "xr-code", mode, vision: visionBlock || undefined });
   850→  }
   851→
   852→  // creation request — build the real asset, then present it in one natural line
   853→  const asset =
   854→    act.action === "image"
   855→      ? await runImageGen((act as { prompt: string }).prompt, (act as { size?: string }).size)
   856→      : await runVideoGen((act as { prompt: string }).prompt, (act as { with_audio?: boolean }).with_audio === true);
   857→  const composerMsgs = [
   858→    ...chatText,
   859→    { role: "assistant" as const, content: JSON.stringify(act) },
   860→    {
   861→      role: "user" as const,
   862→      content: asset
   863→        ? `Tool result: the ${act.action} was generated successfully and is attached in the chat. ` +
   864→          "Write ONE short friendly sentence presenting it to the user — never mention JSON, tools, URLs or engines."
   865→        : `Tool result: the ${act.action} generation failed or timed out this time. ` +
   866→          "Apologize in ONE short natural sentence and suggest trying again in a moment. Never mention JSON, tools or engines.",
   867→    },
   868→  ];
   869→  const composed = await glmChat(SYSTEM_KODER, composerMsgs, engineUsed);
   870→  const text = composed.text ||
   871→    (asset ? "Here it is — freshly generated for you." : "The generation didn't finish this time — give it another try in a moment.");
   872→  return NextResponse.json({
   873→    ok: true,
   874→    text,
   875→    engine: "xr-code",
   876→    mode,
   877→    asset: asset ? { type: act.action, ...asset, prompt: (act as { prompt: string }).prompt } : undefined,
   878→    vision: visionBlock || undefined,
   879→  });
