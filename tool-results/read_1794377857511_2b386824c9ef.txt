   169→// a real directive is a SHORT reply that IS a JSON object — long answers with a

   171→function parseAction(reply: string | null): Action | null {
   172→  if (!reply) return null;
   173→  const s = reply.trim().replace(/^```[a-z]*\s*/i, "").replace(/\s*```$/, "").trim();
   174→  if (s.length > 600 || !s.startsWith("{")) return null;
   175→  try {
   176→    const o = JSON.parse(s) as Record<string, unknown>;
   177→    if (o.action === "search" && typeof o.query === "string" && o.query.trim().length > 1) {
   178→      return { action: "search", query: o.query.trim().slice(0, 200) };
   179→    }
   180→    if (
   181→      (o.action === "image" || o.action === "video") &&
   182→      typeof o.prompt === "string" && o.prompt.trim().length > 2
   183→    ) {
   184→      return {
   185→        action: o.action,
   186→        prompt: o.prompt.trim().slice(0, 800),
   187→        size: typeof o.size === "string" ? o.size : undefined,
   188→        with_audio: o.with_audio === true,
   189→      };
   190→    }
   191→  } catch {
   192→    /* normal text */
   193→  }
   194→  return null;
   195→}

   279→// does this message deserve live web results?
   280→// knowledge / current-info questions — yes. pure code tasks and math/reasoning — no.
   281→function wantsWeb(q: string): boolean {
   282→  const s = q.toLowerCase();
   283→  if (s.length < 4) return false;
   284→  // identity / meta questions about the AI itself — answer directly, never search
   285→  if (/(who are you|what are you|your name|who made you|who created you|who built you|about yourself|who are u)/.test(s)) return false;
   286→  const questionWord = /^(who|what|when|where|why|how|which|whose|is|are|was|were|do|does|did|can|could|should|define|explain|tell me about)\b/.test(s);
   287→  const currentInfo = /(latest|current|today|right now|news|price of|net worth|population|how old|release date|who is|what is)/.test(s);
   288→  const followUp = /^(and|also|his|her|its|their|what about|how about)\b/.test(s) && s.includes("?");
   289→  if (!(questionWord || currentInfo || followUp)) return false;
   290→  // pure code tasks never search, even if phrased oddly long
   291→  if (/^(write|create|make|build|fix|refactor|optimi[sz]e|implement|add)\b/.test(s) && !s.includes("?") && !/who is|what is|latest|current|news/.test(s)) return false;
   292→  return true;
   293→}

   295→// context-aware search query: short follow-ups like "how old is he?" inherit

   297→function searchQuery(safe: Msg[]): string {
   298→  const users = safe.filter(m => m.role === "user");
   299→  const last = (users[users.length - 1]?.content ?? "").trim();
   300→  if (
   301→    users.length >= 2 &&
   302→    (last.length < 45 || /^(and|also|his|her|its|their|what about|how about|then|so)\b/i.test(last))
   303→  ) {
   304→    const prev = users[users.length - 2].content.slice(0, 120);
   305→    return `${prev} ${last}`.slice(0, 300);
   306→  }
   307→  return last.slice(0, 300);
   308→}

   317→const SEARCH_UA =

   320→function decodeEntities(s: string): string {
   321→  return s
   322→    .replace(/&amp;/g, "&").replace(/&lt;/g, "<").replace(/&gt;/g, ">")
   323→    .replace(/&quot;/g, '"').replace(/&#0?39;|&apos;|&#x27;/g, "'")
   324→    .replace(/&nbsp;/g, " ").replace(/&#(\d+);/g, (_, d) => String.fromCharCode(Number(d)));
   325→}
   326→function stripTags(s: string): string {
   327→  return decodeEntities(String(s).replace(/<[^>]*>/g, "")).replace(/\s+/g, " ").trim();
   328→}

   330→async function fetchPage(url: string, ms: number): Promise<string | null> {
   331→  try {
   332→    const ctrl = new AbortController();
   333→    const timer = setTimeout(() => ctrl.abort(), ms);
   334→    const res = await fetch(url, {
   335→      signal: ctrl.signal,
   336→      redirect: "follow",
   337→      headers: { "User-Agent": SEARCH_UA, "Accept-Language": "en-US,en;q=0.9" },
   338→    });
   339→    clearTimeout(timer);
   340→    if (!res.ok) return null;
   341→    return await res.text();
   342→  } catch {
   343→    return null;
   344→  }
   345→}

   347→// Google News RSS — real Google results, no JS wall, no API key
   348→async function googleNews(q: string): Promise<SearchItem[]> {
   349→  const xml = await fetchPage(
   350→    `https://news.google.com/rss/search?q=${encodeURIComponent(q)}&hl=en-US&gl=US&ceid=US:en`, 9000);
   351→  if (!xml) return [];
   352→  const out: SearchItem[] = [];
   353→  for (const chunk of xml.split("<item>").slice(1, 9)) {
   354→    const name = stripTags(chunk.match(/<title>([\s\S]*?)<\/title>/)?.[1] || "");
   355→    const url = (chunk.match(/<link>([\s\S]*?)<\/link>/)?.[1] || "").trim();
   356→    const date = (chunk.match(/<pubDate>([\s\S]*?)<\/pubDate>/)?.[1] || "").trim();
   357→    if (!name || !url) continue;
   358→    out.push({ name, snippet: name, url, date: date ? date.slice(5, 16) : undefined });
   359→  }
   360→  return out;
   361→}

   363→// DuckDuckGo HTML — general-web gap filler (Google News skews to current events)
   364→async function duckWeb(q: string): Promise<SearchItem[]> {
   365→  let html: string | null = null;
   366→  for (let i = 0; i < 2 && !html; i++) {
   367→    html = await fetchPage(`https://html.duckduckgo.com/html/?q=${encodeURIComponent(q)}`, 8000);
   368→  }
   369→  if (!html) return [];
   370→  const out: SearchItem[] = [];
   371→  const re = /<a[^>]*class="[^"]*result__a[^"]*"[^>]*href="([^"]+)"[^>]*>([\s\S]*?)<\/a>/g;
   372→  const snips = [...html.matchAll(/<a[^>]*class="[^"]*result__snippet[^"]*"[^>]*>([\s\S]*?)<\/a>/g)]
   373→    .map(m2 => stripTags(m2[1]));
   374→  let m: RegExpExecArray | null;
   375→  while ((m = re.exec(html)) && out.length < 8) {
   376→    let url = m[1];
   377→    const uddg = url.match(/[?&]uddg=([^&]+)/);
   378→    if (uddg) { try { url = decodeURIComponent(uddg[1]); } catch { /* keep raw */ } }
   379→    const name = stripTags(m[2]);
   380→    if (!name || !/^https?:\/\//.test(url) || /duckduckgo\.com/.test(url)) continue;
   381→    out.push({ name, snippet: snips[out.length] || name, url });
   382→  }
   383→  return out;
   384→}

   408→// merge: Google News first, DuckDuckGo web second, hosted index as the safety net
   409→async function googleSearch(q: string): Promise<SearchItem[]> {
   410→  const [news, web] = await Promise.all([googleNews(q), duckWeb(q)]);
   411→  const seen = new Set<string>();
   412→  const merged: SearchItem[] = [];
   413→  for (const it of [...news, ...web]) {
   414→    const key = it.url.replace(/[?#].*$/, "").toLowerCase();
   415→    if (!it.url || seen.has(key)) continue;
   416→    seen.add(key);
   417→    merged.push(it);
   418→    if (merged.length >= 8) break;
   419→  }
   420→  if (merged.length < 3) {
   421→    for (const it of await hostedSearch(q)) {
   422→      const key = it.url.replace(/[?#].*$/, "").toLowerCase();
   423→      if (seen.has(key)) continue;
   424→      seen.add(key);
   425→      merged.push(it);
   426→      if (merged.length >= 8) break;
   427→    }
   428→  }
   429→  return merged;
   430→}

   433→async function googleContext(q: string): Promise<{ block: string; items: SearchItem[] }> {
   434→  const items = await googleSearch(q);
   435→  if (!items.length) return { block: "", items: [] };
   436→  const lines = items.map(
   437→    (it, i) =>
   438→      `${i + 1}. ${it.name}${it.date ? ` (${it.date})` : ""}\n` +
   439→      `   ${String(it.snippet || "").slice(0, 500)}\n` +
   440→      `   source: ${it.url}`
   441→  );
   442→  const block =
   443→    "\n\nLIVE WEB RESULTS (Google search) for this question:\n" +
   444→    lines.join("\n") +
   445→    "\n\nAnswering rules: base your reply on these results. Give the answer the way the " +
   446→    "top result would read — same facts, but with light grammar and wording changes so it is " +
   447→    "not a verbatim copy. If the results do not actually answer the question, answer from your " +
   448→    "own knowledge instead. Be direct and concise, no markdown headings. " +
   449→    "Do not write a Sources line.";

   514→// Anthropic-style /v1/messages caller. Returns text or null (null = any

   516→async function primeChat(sys: string, msgs: { role: "user" | "assistant"; content: string }[], model: string): Promise<string | null> {
   517→  if (!PRIME_BASE || !PRIME_KEY) return null;
   518→  try {
   519→    // Anthropic messages must alternate roles and start with a user turn.
   520→    const clean: { role: "user" | "assistant"; content: string }[] = [];
   521→    for (const m of msgs) {
   522→      const last = clean[clean.length - 1];
   523→      if (last && last.role === m.role) last.content += "\n\n" + m.content;
   524→      else clean.push({ role: m.role, content: m.content });
   525→    }
   526→    while (clean.length && clean[0].role !== "user") clean.shift();
   527→    if (!clean.length) return null;
   528→    // the gateway injects its own "you are Prime AI" system prompt ahead of
   529→    // ours — re-anchor the identity on the first user turn as well
   530→    clean[0] = {
   531→      role: "user",
   532→      content:
   533→        "(Reminder: you are XR Code, the AI of the XR-Code app by YUX FF. " +
   534→        "Never claim any other identity or mention the underlying model/provider.)\n\n" +
   535→        clean[0].content,
   536→    };
   537→    const res = (await Promise.race([
   538→      fetch(PRIME_BASE + "/v1/messages", {
   539→        method: "POST",
   540→        headers: { "Content-Type": "application/json", Authorization: `Bearer ${PRIME_KEY}` },
   541→        body: JSON.stringify({ model, max_tokens: 4096, system: sys, messages: clean }),
   542→        signal: AbortSignal.timeout(45000),
   543→      }),
   544→      new Promise<null>(r => setTimeout(() => r(null), 50000)),
   545→    ])) as Response | null;
   546→    if (!res || !res.ok) return null;
   547→    const d = (await res.json()) as { content?: { type?: string; text?: string }[] };
   548→    const parts = Array.isArray(d?.content) ? d.content : [];

   550→      .filter(p => p?.type === "text")
   551→      .map(p => String(p.text || ""))
   552→      .join("")

   563→async function glmChat(sys: string, msgs: { role: "user" | "assistant"; content: string }[], model: string): Promise<ChatResult> {
   564→  // self-healing: up to 2 attempts, re-creating the engine client each time —
   565→  // a stale/poisoned SDK instance (env refresh after recycle) recovers instantly.
   566→  // 429 short-circuit: return quota:true at once — the CALLER decides whether
   567→  // to probe the other free lanes (FALLBACK_MODELS) or answer from Google /
   568→  // honest outage. No unbounded retries here: one message must never spin
   569→  // 16+ gateway calls.
   570→  for (let attempt = 0; attempt < 2; attempt++) {
   571→    try {
   572→      const zai = await ZAI.create();
   573→      // thinking toggle is a GLM knob — applied to glm-* models only, never
   574→      // to the emergency dpsk/argon lanes
   575→      const glmish = model.startsWith("glm");
   576→      const res = (await Promise.race([
   577→        zai.chat.completions.create({
   578→          model,
   579→          messages: [{ role: "system", content: sys }, ...msgs],
   580→          ...(glmish ? { thinking: { type: "disabled" as const } } : {}),
   581→        }),
   582→        new Promise<null>(r => setTimeout(() => r(null), 60000)),
   583→      ])) as { choices?: { message?: { content?: string } }[] } | null;
   584→      const text = String(res?.choices?.[0]?.message?.content || "").trim();
   585→      if (text) return { text, quota: false };
   586→      // empty reply is an engine hiccup, not a quota signal — retry once
   587→    } catch (e) {
   588→      const msg = String((e as Error)?.message || e);
   589→      if (/\b429\b|too many requests/i.test(msg)) return { text: null, quota: true };
   590→      // engine hiccup — fall through to retry
   591→    }
   592→    if (attempt < 1) await new Promise(r => setTimeout(r, 1200));
   593→  }
   594→  return { text: null, quota: false }; // caller tries the next engine
   595→}

   612→// pull plain text out of string-or-multipart content
   613→function textOf(content: unknown): string {
   614→  if (typeof content === "string") return content;
   615→  if (Array.isArray(content)) {
   616→    return content
   617→      .map(p => (p && typeof p === "object" && typeof (p as Part).text === "string" ? ((p as Part).text as string) : ""))
   618→      .join("\n");
   619→  }
   620→  return "";
   621→}

   623→// ---- real vision preprocessing ----
   624→// pull base64 image blocks (photos + video keyframes) out of the last user message
   625→function imagesOf(content: unknown): { mt: string; b64: string }[] {
   626→  if (!Array.isArray(content)) return [];
   627→  const out: { mt: string; b64: string }[] = [];
   628→  for (const p of content) {
   629→    const part = p as Part;
   630→    const src = part?.source;
   631→    if (part?.type === "image" && src && typeof src.data === "string" && src.data.length > 64) {
   632→      const mt =
   633→        src.media_type && /^image\/(png|jpe?g|webp|gif)$/.test(src.media_type)
   634→          ? (src.media_type === "image/jpg" ? "image/jpeg" : src.media_type)
   635→          : "image/jpeg";
   636→      if (src.data.length <= 4_200_000) out.push({ mt, b64: src.data });
   637→    }
   638→  }
   639→  return out.slice(0, 8);
   640→}

   820→              `Tool result: the ${a.action} was generated successfully and is attached in the chat. ` +
   821→              "Write your FINAL answer now: one short friendly sentence presenting it (plus anything else the task needed). " +
   822→              "Never mention JSON, tools, URLs or engines.",
   823→          },
   824→        ];
   825→        const composed = await glmChat(sys, composerMsgs, agentModel);
   826→        const prefix = trace.length ? `Agent · ${trace.join(" → ")}\n\n` : "";
   827→        return NextResponse.json({
   828→          ok: true,
   829→          text: prefix + (composed.text || (a.action === "image" ? "Here it is — freshly generated for you." : "Here's your clip — freshly generated for you.")),
   830→          engine: "xr-code",
   831→          mode,
   832→          asset: { type: a.action, ...asset, prompt: (a as { prompt: string }).prompt },
   833→          trace,
   834→          vision: visionBlock || undefined,
   835→        });
   836→      }
   837→      chatText.push({ role: "assistant", content: JSON.stringify(a) });
   838→      chatText.push({
   839→        role: "user",
   840→        content: `Tool result: the ${a.action} generation failed or timed out. Continue the task another way or deliver the best answer you can without it.`,
   841→      });
   842→    }
   843→    // budget spent — force the final answer
   844→    chatText.push({ role: "user", content: "TOOL BUDGET REACHED — deliver your final answer NOW, no more tool calls." });
   845→    const last = await glmChat(sys, chatText, agentModel);
   846→    const prefix = trace.length ? `Agent · ${trace.join(" → ")}\n\n` : "";
   847→    if (last.text) {
   848→      return NextResponse.json({ ok: true, text: prefix + last.text, engine: "xr-code", mode, trace, vision: visionBlock || undefined });
   849→    }
   850→    return NextResponse.json({ ok: false, error: "agent ran out of steps — try again" });
   851→  }
   852→
   853→  // one brain first: GLM (the original first model) — plain replies return;
   854→  // action directives run the tools. On a 429 the chain walks the emergency
   855→  // free lanes (dpsk-4.1-flash, dpsk-4.1, glm-4.6, argon-1.0) before giving up.
   856→  // Quota storm policy: a 429 means the primary's quota bucket is exhausted.
   857→  // One 3s grace retry on the primary, then walk the emergency free lanes
   858→  // (each one quick attempt — a dry lane 429s in ~100ms). If every lane is
   859→  // dry: answer with an honest outage bubble (outageReply) instead of
   860→  // ok:false — the obfuscated client's failure branch would show a lying
   861→  // "unreachable" message. QnA and Koder never go dark: with live Google
   862→  // results in hand they answer straight from the search engine. Error text
   863→  // must never match the client's /engine/i auto-retry regex, or one message
   864→  // would spin 5 retries.
   865→  // the Google fallback answers for BOTH QnA and Koder when the GLM quota is
   866→  // dead and live search results are already in hand — the chat never goes dark
   867→  const googleOutage = (): NextResponse | null =>
   868→    (mode === "qna" || mode === "koder") && lastSearch.length
   869→      ? NextResponse.json({
   870→          ok: true,
   871→          text: withSources(googleFallback(searchQuery(safe), lastSearch), mode, lastSearch),
   872→          engine: "google-search",
   873→          mode,
   874→          vision: visionBlock || undefined,
   875→        })
   876→      : null;
   877→  // Prime-selected requests: the owner's gateway answers FIRST; if it is
   878→  // unreachable / slow / dry, the Z.ai brain takes over (kimi + emergency
   879→  // lanes) — the chat never goes dark. Z.ai-selected requests: the gateway
   880→  // rides LAST as an extra emergency lane (separate quota bucket) before the
   881→  // honest outage bubble.
   882→  const chain = PRIME_MODELS.has(chosen)
   883→    ? [chosen, ...FALLBACK_MODELS]
   884→    : [chosen, ...FALLBACK_MODELS.filter(m => m !== chosen), PRIME_FLASH];
   885→  // one lane call, two transports: prime ids ride the Anthropic-style gateway,
   886→  // everything else rides the ZAI SDK
   887→  const laneCall = async (engine: string) =>
   888→    PRIME_MODELS.has(engine)
   889→      ? { text: await primeChat(sys, chatText, engine), quota: false }
   890→      : await glmChat(sys, chatText, engine);
   891→  let actText: string | null = null;
   892→  let engineUsed = "";
   893→  let act: Action | null = null;
   894→  for (const engine of chain) {
   895→    const r = await laneCall(engine);
   896→    if (r.quota) {
   897→      if (engine !== chosen) continue; // fallback lane dry — next lane
   898→      // primary is throttled: one grace retry, then the remaining lanes
   899→      await new Promise(r2 => setTimeout(r2, 3000));
   900→      const retry = await glmChat(sys, chatText, chosen);
   901→      if (retry.text) {
   902→        const a2 = parseAction(retry.text);
   903→        if (a2) { actText = retry.text; engineUsed = chosen; act = a2; break; }
   904→        return NextResponse.json({ ok: true, text: withSources(retry.text, mode, lastSearch), engine: "xr-code", mode, vision: visionBlock || undefined });
   905→      }
   906→      continue;
   907→    }
   908→    if (!r.text) continue;
   909→    const t = r.text;
   910→    const a = parseAction(t);
   911→    if (a) {
   912→      actText = t;
   913→      engineUsed = engine;
   914→      act = a;
   915→      break;
   916→    }
   917→    return NextResponse.json({ ok: true, text: withSources(t, mode, lastSearch), engine: "xr-code", mode, vision: visionBlock || undefined });
   918→  }
   919→  if (!actText || !act) {
   920→    const out = googleOutage();
   921→    if (out) return out;
   922→    return NextResponse.json({ ok: true, text: outageReply(mode), engine: "xr-code", mode, vision: visionBlock || undefined });
   923→  }
   924→
   925→  // creation request — build the real asset, then present it in one natural line
   926→  const asset =
   927→    act.action === "image"
   928→      ? await runImageGen((act as { prompt: string }).prompt, (act as { size?: string }).size)
   929→      : await runVideoGen((act as { prompt: string }).prompt, (act as { with_audio?: boolean }).with_audio === true);
   930→  const composerMsgs = [
   931→    ...chatText,
   932→    { role: "assistant" as const, content: JSON.stringify(act) },
   933→    {
   934→      role: "user" as const,
   935→      content: asset
   936→        ? `Tool result: the ${act.action} was generated successfully and is attached in the chat. ` +
   937→          "Write ONE short friendly sentence presenting it to the user — never mention JSON, tools, URLs or engines."
   938→        : `Tool result: the ${act.action} generation failed or timed out this time. ` +
   939→          "Apologize in ONE short natural sentence and suggest trying again in a moment. Never mention JSON, tools or engines.",
   940→    },
   941→  ];
   942→  const composed = PRIME_MODELS.has(engineUsed)
   943→    ? { text: await primeChat(SYSTEM_KODER, composerMsgs, engineUsed) }
   944→    : await glmChat(SYSTEM_KODER, composerMsgs, engineUsed);
   945→  const text = composed.text ||
   946→    (asset ? "Here it is — freshly generated for you." : "The generation didn't finish this time — give it another try in a moment.");
   947→  return NextResponse.json({
   948→    ok: true,
   949→    text,
   950→    engine: "xr-code",
   951→    mode,
   952→    asset: asset ? { type: act.action, ...asset, prompt: (act as { prompt: string }).prompt } : undefined,
   953→    vision: visionBlock || undefined,
   954→  });
