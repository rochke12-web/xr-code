   156→// a real directive is a SHORT reply that IS a JSON object — long answers with a
   157→// JSON snippet inside are just code, never a tool call
   158→function parseAction(reply: string | null): Action | null {
   159→  if (!reply) return null;
   160→  const s = reply.trim().replace(/^```[a-z]*\s*/i, "").replace(/\s*```$/, "").trim();
   161→  if (s.length > 600 || !s.startsWith("{")) return null;
   162→  try {
   163→    const o = JSON.parse(s) as Record<string, unknown>;
   164→    if (o.action === "search" && typeof o.query === "string" && o.query.trim().length > 1) {
   165→      return { action: "search", query: o.query.trim().slice(0, 200) };
   166→    }
   167→    if (
   168→      (o.action === "image" || o.action === "video") &&
   169→      typeof o.prompt === "string" && o.prompt.trim().length > 2
   170→    ) {
   171→      return {
   172→        action: o.action,
   173→        prompt: o.prompt.trim().slice(0, 800),
   174→        size: typeof o.size === "string" ? o.size : undefined,
   175→        with_audio: o.with_audio === true,
   176→      };
   177→    }
   178→  } catch {
   179→    /* normal text */
   180→  }
   181→  return null;
   182→}

   266→// does this message deserve live web results?
   267→// knowledge / current-info questions — yes. pure code tasks and math/reasoning — no.
   268→function wantsWeb(q: string): boolean {
   269→  const s = q.toLowerCase();
   270→  if (s.length < 4) return false;
   271→  // identity / meta questions about the AI itself — answer directly, never search
   272→  if (/(who are you|what are you|your name|who made you|who created you|who built you|about yourself|who are u)/.test(s)) return false;
   273→  const questionWord = /^(who|what|when|where|why|how|which|whose|is|are|was|were|do|does|did|can|could|should|define|explain|tell me about)\b/.test(s);
   274→  const currentInfo = /(latest|current|today|right now|news|price of|net worth|population|how old|release date|who is|what is)/.test(s);
   275→  const followUp = /^(and|also|his|her|its|their|what about|how about)\b/.test(s) && s.includes("?");
   276→  if (!(questionWord || currentInfo || followUp)) return false;
   277→  // pure code tasks never search, even if phrased oddly long
   278→  if (/^(write|create|make|build|fix|refactor|optimi[sz]e|implement|add)\b/.test(s) && !s.includes("?") && !/who is|what is|latest|current|news/.test(s)) return false;
   279→  return true;
   280→}

   282→// context-aware search query: short follow-ups like "how old is he?" inherit
   283→// the previous question's topic so the search still finds the right thing.
   284→function searchQuery(safe: Msg[]): string {
   285→  const users = safe.filter(m => m.role === "user");
   286→  const last = (users[users.length - 1]?.content ?? "").trim();
   287→  if (
   288→    users.length >= 2 &&
   289→    (last.length < 45 || /^(and|also|his|her|its|their|what about|how about|then|so)\b/i.test(last))
   290→  ) {
   291→    const prev = users[users.length - 2].content.slice(0, 120);
   292→    return `${prev} ${last}`.slice(0, 300);
   293→  }
   294→  return last.slice(0, 300);
   295→}

   385→async function glmChat(sys: string, msgs: { role: "user" | "assistant"; content: string }[], model: string): Promise<ChatResult> {
   386→  // self-healing: up to 2 attempts, re-creating the engine client each time —
   387→  // a stale/poisoned SDK instance (env refresh after recycle) recovers instantly.
   388→  // 429 short-circuit: the whole family shares one key quota — once the gateway
   389→  // says "too many requests" for any member, every other member will too, and
   390→  // walking the rest of the chain only deepens the storm (16+ calls per message).
   391→  // Detect it and bail out at once; the caller decides what to tell the user.
   392→  for (let attempt = 0; attempt < 2; attempt++) {
   393→    try {
   394→      const zai = await ZAI.create();
   395→      // thinking toggle is a GLM-specific knob — don't send it to other
   396→      // families (dpsk/argon may reject unknown params)
   397→      const glmish = model.startsWith("glm");
   398→      const res = (await Promise.race([
   399→        zai.chat.completions.create({
   400→          model,
   401→          messages: [{ role: "system", content: sys }, ...msgs],
   402→          ...(glmish ? { thinking: { type: "disabled" as const } } : {}),
   403→        }),
   404→        new Promise<null>(r => setTimeout(() => r(null), 60000)),
   405→      ])) as { choices?: { message?: { content?: string } }[] } | null;
   406→      const text = String(res?.choices?.[0]?.message?.content || "").trim();
   407→      if (text) return { text, quota: false };
   408→      // empty reply is an engine hiccup, not a quota signal — retry once
   409→    } catch (e) {
   410→      const msg = String((e as Error)?.message || e);
   411→      if (/\b429\b|too many requests/i.test(msg)) return { text: null, quota: true };
   412→      // engine hiccup — fall through to retry
   413→    }
   414→    if (attempt < 1) await new Promise(r => setTimeout(r, 1200));
   415→  }
   416→  return { text: null, quota: false }; // caller tries the next engine
   417→}

   422→const RATE_LIMITED =

   426→// pull plain text out of string-or-multipart content
   427→function textOf(content: unknown): string {
   428→  if (typeof content === "string") return content;
   429→  if (Array.isArray(content)) {
   430→    return content
   431→      .map(p => (p && typeof p === "object" && typeof (p as Part).text === "string" ? ((p as Part).text as string) : ""))
   432→      .join("\n");
   433→  }
   434→  return "";
   435→}
   436→
   437→// ---- real vision preprocessing ----
   438→// pull base64 image blocks (photos + video keyframes) out of the last user message
   439→function imagesOf(content: unknown): { mt: string; b64: string }[] {
   440→  if (!Array.isArray(content)) return [];
   441→  const out: { mt: string; b64: string }[] = [];
   442→  for (const p of content) {
   443→    const part = p as Part;
   444→    const src = part?.source;
   445→    if (part?.type === "image" && src && typeof src.data === "string" && src.data.length > 64) {
   446→      const mt =
   447→        src.media_type && /^image\/(png|jpe?g|webp|gif)$/.test(src.media_type)
   448→          ? (src.media_type === "image/jpg" ? "image/jpeg" : src.media_type)
   449→          : "image/jpeg";
   450→      if (src.data.length <= 4_200_000) out.push({ mt, b64: src.data });
   451→    }
   452→  }
   453→  return out.slice(0, 8);
   454→}

   456→// describe the images with a vision model; returns one 'Image N: …' line each
   457→async function describeImages(imgs: { mt: string; b64: string }[]): Promise<string> {
   458→  if (!imgs.length) return "";
   459→  try {
   460→    const zai = await ZAI.create();
   461→    const content: Extract<VisionMessage["content"], unknown[]> = [
   462→      {
   463→        type: "text",
   464→        text:
   465→          "You are the vision preprocessor of a chat app. For EACH attached image, in order, " +
   466→          "write exactly one line: 'Image N: <description>'. Describe concretely what is shown " +
   467→          "(objects, colors, scene, layout) and transcribe ALL visible text verbatim. " +
   468→          "No preamble, no extra lines.",
   469→      },
   470→    ];
   471→    for (const im of imgs) {
   472→      content.push({ type: "image_url", image_url: { url: `data:${im.mt};base64,${im.b64}` } });
   473→    }
   474→    // SDK types demand `model` and omit `role`, but the verified-working runtime
   475→    // payload needs neither — cast instead of changing what works.
   476→    const visionBody = {
   477→      messages: [{ role: "user", content }],
   478→      thinking: { type: "disabled" as const },
   479→    };
   480→    const res = (await Promise.race([
   481→      zai.chat.completions.createVision(
   482→        visionBody as unknown as Parameters<typeof zai.chat.completions.createVision>[0]
   483→      ),
   484→      new Promise<null>(r => setTimeout(() => r(null), 32000)),
   485→    ])) as { choices?: { message?: { content?: string } }[] } | null;
   486→    return String(res?.choices?.[0]?.message?.content || "").trim();
   487→  } catch {
   488→    return ""; // vision preprocessing down → fall back to blind text notes
   489→  }
   490→}

   496→export async function POST(req: Request) {
   497→  let body: { messages?: RawMsg[]; model?: string; mode?: string } | null = null;
   498→  try {
   499→    body = await req.json();
   500→  } catch {
   501→    /* no / invalid body */
   502→  }
   503→  const raw = Array.isArray(body?.messages) ? (body!.messages as RawMsg[]) : null;
   504→  const safe: Msg[] = raw
   505→    ? raw
   506→        .slice(-24)
   507→        .filter(m => m && (m.role === "user" || m.role === "assistant") && m.content != null)
   508→        .map(m => ({ role: m.role as "user" | "assistant", content: textOf(m.content).slice(0, 8000) }))
   509→    : [];
   510→  if (!safe.length) {
   511→    return NextResponse.json({ ok: false, error: "bad request" }, { status: 400 });
   512→  }
   513→
   514→  // two modes: Koder (developer brain) and QnA (web-grounded answers, sources stay hidden)
   515→  const modeRaw = typeof body?.mode === "string" ? body.mode.trim().toLowerCase() : "";
   516→  const mode = modeRaw === "qna" || modeRaw === "sight" || modeRaw === "agent" ? modeRaw : "koder";
   517→  let sys =
   518→    (mode === "qna" ? SYSTEM_QNA : mode === "sight" ? SYSTEM_SIGHT : mode === "agent" ? SYSTEM_AGENT : SYSTEM_KODER) +
   519→    TOOL_RULES +
   520→    (mode === "agent" ? AGENT_SEARCH_RULE : "");
   521→  const lastUser = [...safe].reverse().find(m => m.role === "user");
   522→  // one brain, all GLM: explicit family member is honored, otherwise auto-route
   523→  const wanted = typeof body?.model === "string" ? body.model.trim() : "";
   524→  const chosen = GLM_MODELS.has(wanted) ? wanted : autoEngine(lastUser?.content || "", safe);
   525→  // the user is talking about THEIR attachment — web search would only distract.
   526→  // QnA mode grounds almost every real question in live results (sources never shown).
   527→  // Agent mode skips auto-grounding: it searches itself, on demand, per step.
   528→  if (mode !== "agent" && lastUser && !/\[attached (image|video|file|audio):/i.test(lastUser.content)) {
   529→    const q = lastUser.content.toLowerCase();
   530→    const identityQ = /(who are you|what are you|your name|who made you|who created you|who built you|about yourself|who are u)/.test(q);
   531→    const looksQuestion = q.length >= 4 && (/\?/.test(q) || /^(who|what|when|where|why|how|which|whose|is|are|was|were|do|does|did|can|could|should|define|explain|tell me)\b/.test(q) || q.length > 60);
   532→    const forceWeb = mode === "qna" && !identityQ && looksQuestion;
   533→    if (forceWeb || wantsWeb(lastUser.content)) {
   534→      sys += await googleContext(searchQuery(safe));
   535→      if (mode === "qna") {
   536→        sys += "\n\nSECRECY REMINDER: answer from the LIVE WEB RESULTS as your own knowledge — never mention sources, " +
   537→               "sites, links or that anything was searched. No Sources line, even if the user asks for one.";
   538→      } else if (/\b(sources?|cite|citations?|references?|links?|urls?)\b/i.test(lastUser.content)) {
   539→        sys += "\n\nThe user explicitly asked for sources: end your reply with a final line 'Sources:' " +
   540→               "followed by up to 3 of the LIVE WEB RESULTS URLs above.";
   541→      }
   542→    }
   543→  }
   544→
   545→  // ---- real vision: describe the attached photos / video keyframes, then feed
   546→  // the description to the text brain as a [VISION] block ----
   547→  let visionBlock = "";
   548→  const rawLastUser = raw ? [...raw].reverse().find(m => m && m.role === "user") : null;
   549→  const imgs = imagesOf(rawLastUser?.content);
   550→  if (imgs.length) {
   551→    const desc = await describeImages(imgs);
   552→    if (desc) {
   553→      visionBlock = "\n\n[VISION — what the attached images show]\n" + desc + "\n[end VISION]";
   554→      sys = sys.replace(ATT_BLIND, ATT_VISION);
   555→    }
   556→  }
   557→
   558→  const chatText = safe.map(m => ({ role: m.role, content: m.content }));
   559→  if (visionBlock) {
   560→    const lu = [...chatText].reverse().find(m => m.role === "user");
   561→    if (lu) lu.content = (lu.content + visionBlock).slice(0, 12000);
   562→  }
   563→
   564→  // ---- AGENT MODE — autonomous multi-step loop ----
   565→  // The model answers OR emits one tool directive per step; the server runs it
   566→  // (live web search / image gen / video gen) and feeds the result back until
   567→  // the task is done or the budget is spent. GLM flagship drives it.
   568→  if (mode === "agent") {
   569→    const MAX_STEPS = 4;
   570→    const trace: string[] = [];
   571→    const agentModel = GLM_FLAGSHIP;
   572→    let engineOut = agentModel;
   573→    for (let step = 0; step < MAX_STEPS; step++) {
   574→      const r = await glmChat(sys, chatText, agentModel);
   575→      if (r.quota) {
   576→        await new Promise(r2 => setTimeout(r2, 3000));
   577→        const retry = await glmChat(sys, chatText, DPSK_QUICK);
   578→        if (retry.text) {
   579→          return NextResponse.json({ ok: true, text: retry.text, engine: DPSK_QUICK, mode, vision: visionBlock || undefined });
   580→        }
   581→        return NextResponse.json({ ok: false, error: RATE_LIMITED });
   582→      }
   583→      if (!r.text) {
   584→        // one cross-family rescue, then give up honestly
   585→        const rescue = await glmChat(sys, chatText, DPSK_FLAGSHIP);
   586→        if (rescue.text) {
   587→          return NextResponse.json({ ok: true, text: rescue.text, engine: DPSK_FLAGSHIP, mode, vision: visionBlock || undefined });
   588→        }
   589→        return NextResponse.json({ ok: false, error: "agent stalled — try again in a moment" });
   590→      }
   591→      const a = parseAction(r.text);
   592→      if (!a) {
   593→        // final answer delivered
   594→        const prefix = trace.length ? `Agent · ${trace.join(" → ")}\n\n` : "";
   595→        return NextResponse.json({ ok: true, text: prefix + r.text, engine: engineOut, mode, trace, vision: visionBlock || undefined });
   596→      }
   597→      if (a.action === "search") {
   598→        trace.push(`search "${a.query}"`);
   599→        const ctx = (await googleContext(a.query)) || "";
   600→        chatText.push({ role: "assistant", content: JSON.stringify(a) });
   601→        chatText.push({
   602→          role: "user",
   603→          content:
   604→            `TOOL RESULT — live web results for "${a.query}":\n` +
   605→            (ctx ? ctx.slice(0, 6000) : "(no results found — answer from your own knowledge)"),
   606→        });
   607→        continue;
   608→      }
   609→      // image / video generation
   610→      trace.push(`${a.action} ✓`);
   611→      const asset =
   612→        a.action === "image"
   613→          ? await runImageGen((a as { prompt: string }).prompt, (a as { size?: string }).size)
   614→          : await runVideoGen((a as { prompt: string }).prompt, (a as { with_audio?: boolean }).with_audio === true);
   615→      if (asset) {
   616→        const composerMsgs = [
   617→          ...chatText,
   618→          { role: "assistant" as const, content: JSON.stringify(a) },
   619→          {
   620→            role: "user" as const,
   621→            content:
   622→              `Tool result: the ${a.action} was generated successfully and is attached in the chat. ` +
   623→              "Write your FINAL answer now: one short friendly sentence presenting it (plus anything else the task needed). " +
   624→              "Never mention JSON, tools, URLs or engines.",
   625→          },
   626→        ];
   627→        const composed = await glmChat(sys, composerMsgs, agentModel);
   628→        const prefix = trace.length ? `Agent · ${trace.join(" → ")}\n\n` : "";
   629→        return NextResponse.json({
   630→          ok: true,
   631→          text: prefix + (composed.text || (a.action === "image" ? "Here it is — freshly generated for you." : "Here's your clip — freshly generated for you.")),
   632→          engine: engineOut,
   633→          mode,
   634→          asset: { type: a.action, ...asset, prompt: (a as { prompt: string }).prompt },
   635→          trace,
   636→          vision: visionBlock || undefined,
   637→        });
   638→      }
   639→      chatText.push({ role: "assistant", content: JSON.stringify(a) });
   640→      chatText.push({
   641→        role: "user",
   642→        content: `Tool result: the ${a.action} generation failed or timed out. Continue the task another way or deliver the best answer you can without it.`,
   643→      });
   644→    }
   645→    // budget spent — force the final answer
   646→    chatText.push({ role: "user", content: "TOOL BUDGET REACHED — deliver your final answer NOW, no more tool calls." });
   647→    const last = await glmChat(sys, chatText, agentModel);
   648→    const prefix = trace.length ? `Agent · ${trace.join(" → ")}\n\n` : "";
   649→    if (last.text) {
   650→      return NextResponse.json({ ok: true, text: prefix + last.text, engine: engineOut, mode, trace, vision: visionBlock || undefined });
   651→    }
   652→    return NextResponse.json({ ok: false, error: "agent ran out of steps — try again" });
   653→  }
   654→
   655→  // one GLM brain — plain replies return; action directives run the tools.
   656→  // Cross-family escape hatch appended: if every GLM member is down (non-quota
   657→  // hiccups), argon / dpsk-4.1-flash get a chance before we give up.
   658→  // Quota storm policy: a 429 on ANY member means the shared key is exhausted —
   659→  // short-circuit the whole chain (one grace retry on the quick model after 3s
   660→  // to ride out momentary spikes), then fail fast with a friendly message.
   661→  // Failing fast keeps one user message ≈ 2-3 upstream calls instead of ~80,
   662→  // which is what lets the quota window actually recover.
   663→  // Every AI in one chain, GLM first (yesterday-night order): glm pair →
   664→  // deepseek pair → argon tail.
   665→  const family = [...GLM_FALLBACKS, ...DPSK_FALLBACKS, ...EXTRA_FALLBACKS];
   666→  const chain = [chosen, ...family.filter(m => m !== chosen)];
   667→  let actText: string | null = null;
   668→  let engineUsed = "";
   669→  let act: Action | null = null;
   670→  for (const engine of chain) {
   671→    const r = await glmChat(sys, chatText, engine);
   672→    if (r.quota) {
   673→      await new Promise(r2 => setTimeout(r2, 3000));
   674→      const retry = await glmChat(sys, chatText, DPSK_QUICK);
   675→      if (retry.text) {
   676→        const a2 = parseAction(retry.text);
   677→        if (a2) { actText = retry.text; engineUsed = DPSK_QUICK; act = a2; break; }
   678→        return NextResponse.json({ ok: true, text: retry.text, engine: DPSK_QUICK, mode, vision: visionBlock || undefined });
   679→      }
   680→      return NextResponse.json({ ok: false, error: RATE_LIMITED });
   681→    }
   682→    if (!r.text) continue;
   683→    const t = r.text;
   684→    const a = parseAction(t);
   685→    if (a) {
   686→      actText = t;
   687→      engineUsed = engine;
   688→      act = a;
   689→      break;
   690→    }
   691→    return NextResponse.json({ ok: true, text: t, engine, mode, vision: visionBlock || undefined });
   692→  }
   693→  if (!actText || !act)
   694→    return NextResponse.json({ ok: false, error: "engine unreachable" });
   695→
   696→  // creation request — build the real asset, then present it in one natural line
   697→  const asset =
   698→    act.action === "image"
   699→      ? await runImageGen(act.prompt, act.size)
   700→      : await runVideoGen(act.prompt, act.with_audio === true);
   701→  const composerMsgs = [
   702→    ...chatText,
   703→    { role: "assistant" as const, content: JSON.stringify(act) },
   704→    {
   705→      role: "user" as const,
   706→      content: asset
   707→        ? `Tool result: the ${act.action} was generated successfully and is attached in the chat. ` +
   708→          "Write ONE short friendly sentence presenting it to the user — never mention JSON, tools, URLs or engines."
   709→        : `Tool result: the ${act.action} generation failed or timed out this time. ` +
   710→          "Apologize in ONE short natural sentence and suggest trying again in a moment. Never mention JSON, tools or engines.",
   711→    },
   712→  ];
   713→  const composed = await glmChat(SYSTEM_KODER, composerMsgs, engineUsed);
   714→  const text = composed.text ||
   715→    (asset ? "Here it is — freshly generated for you." : "The generation didn't finish this time — give it another try in a moment.");
   716→  return NextResponse.json({
   717→    ok: true,
   718→    text,
   719→    engine: engineUsed,
   720→    mode,
   721→    asset: asset ? { type: act.action, ...asset, prompt: act.prompt } : undefined,
   722→    vision: visionBlock || undefined,
   723→  });
   724→}
