   157→// a real directive is a SHORT reply that IS a JSON object — long answers with a
   158→// JSON snippet inside are just code, never a tool call
   159→function parseAction(reply: string | null): Action | null {
   160→  if (!reply) return null;
   161→  const s = reply.trim().replace(/^```[a-z]*\s*/i, "").replace(/\s*```$/, "").trim();
   162→  if (s.length > 600 || !s.startsWith("{")) return null;
   163→  try {
   164→    const o = JSON.parse(s) as Record<string, unknown>;
   165→    if (o.action === "search" && typeof o.query === "string" && o.query.trim().length > 1) {
   166→      return { action: "search", query: o.query.trim().slice(0, 200) };
   167→    }
   168→    if (
   169→      (o.action === "image" || o.action === "video") &&
   170→      typeof o.prompt === "string" && o.prompt.trim().length > 2
   171→    ) {
   172→      return {
   173→        action: o.action,
   174→        prompt: o.prompt.trim().slice(0, 800),
   175→        size: typeof o.size === "string" ? o.size : undefined,
   176→        with_audio: o.with_audio === true,
   177→      };
   178→    }
   179→  } catch {
   180→    /* normal text */
   181→  }
   182→  return null;
   183→}

   267→// does this message deserve live web results?
   268→// knowledge / current-info questions — yes. pure code tasks and math/reasoning — no.
   269→function wantsWeb(q: string): boolean {
   270→  const s = q.toLowerCase();
   271→  if (s.length < 4) return false;
   272→  // identity / meta questions about the AI itself — answer directly, never search
   273→  if (/(who are you|what are you|your name|who made you|who created you|who built you|about yourself|who are u)/.test(s)) return false;
   274→  const questionWord = /^(who|what|when|where|why|how|which|whose|is|are|was|were|do|does|did|can|could|should|define|explain|tell me about)\b/.test(s);
   275→  const currentInfo = /(latest|current|today|right now|news|price of|net worth|population|how old|release date|who is|what is)/.test(s);
   276→  const followUp = /^(and|also|his|her|its|their|what about|how about)\b/.test(s) && s.includes("?");
   277→  if (!(questionWord || currentInfo || followUp)) return false;
   278→  // pure code tasks never search, even if phrased oddly long
   279→  if (/^(write|create|make|build|fix|refactor|optimi[sz]e|implement|add)\b/.test(s) && !s.includes("?") && !/who is|what is|latest|current|news/.test(s)) return false;
   280→  return true;
   281→}

   283→// context-aware search query: short follow-ups like "how old is he?" inherit
   284→// the previous question's topic so the search still finds the right thing.
   285→function searchQuery(safe: Msg[]): string {
   286→  const users = safe.filter(m => m.role === "user");
   287→  const last = (users[users.length - 1]?.content ?? "").trim();
   288→  if (
   289→    users.length >= 2 &&
   290→    (last.length < 45 || /^(and|also|his|her|its|their|what about|how about|then|so)\b/i.test(last))
   291→  ) {
   292→    const prev = users[users.length - 2].content.slice(0, 120);
   293→    return `${prev} ${last}`.slice(0, 300);
   294→  }
   295→  return last.slice(0, 300);
   296→}

   305→const SEARCH_UA =

   308→function decodeEntities(s: string): string {
   309→  return s
   310→    .replace(/&amp;/g, "&").replace(/&lt;/g, "<").replace(/&gt;/g, ">")
   311→    .replace(/&quot;/g, '"').replace(/&#0?39;|&apos;|&#x27;/g, "'")
   312→    .replace(/&nbsp;/g, " ").replace(/&#(\d+);/g, (_, d) => String.fromCharCode(Number(d)));
   313→}
   314→function stripTags(s: string): string {
   315→  return decodeEntities(String(s).replace(/<[^>]*>/g, "")).replace(/\s+/g, " ").trim();
   316→}

   318→async function fetchPage(url: string, ms: number): Promise<string | null> {
   319→  try {
   320→    const ctrl = new AbortController();
   321→    const timer = setTimeout(() => ctrl.abort(), ms);
   322→    const res = await fetch(url, {
   323→      signal: ctrl.signal,
   324→      redirect: "follow",
   325→      headers: { "User-Agent": SEARCH_UA, "Accept-Language": "en-US,en;q=0.9" },
   326→    });
   327→    clearTimeout(timer);
   328→    if (!res.ok) return null;
   329→    return await res.text();
   330→  } catch {
   331→    return null;
   332→  }
   333→}

   335→// Google News RSS — real Google results, no JS wall, no API key
   336→async function googleNews(q: string): Promise<SearchItem[]> {
   337→  const xml = await fetchPage(
   338→    `https://news.google.com/rss/search?q=${encodeURIComponent(q)}&hl=en-US&gl=US&ceid=US:en`, 9000);
   339→  if (!xml) return [];
   340→  const out: SearchItem[] = [];
   341→  for (const chunk of xml.split("<item>").slice(1, 9)) {
   342→    const name = stripTags(chunk.match(/<title>([\s\S]*?)<\/title>/)?.[1] || "");
   343→    const url = (chunk.match(/<link>([\s\S]*?)<\/link>/)?.[1] || "").trim();
   344→    const date = (chunk.match(/<pubDate>([\s\S]*?)<\/pubDate>/)?.[1] || "").trim();
   345→    if (!name || !url) continue;
   346→    out.push({ name, snippet: name, url, date: date ? date.slice(5, 16) : undefined });
   347→  }
   348→  return out;
   349→}

   351→// DuckDuckGo HTML — general-web gap filler (Google News skews to current events)
   352→async function duckWeb(q: string): Promise<SearchItem[]> {
   353→  let html: string | null = null;
   354→  for (let i = 0; i < 2 && !html; i++) {
   355→    html = await fetchPage(`https://html.duckduckgo.com/html/?q=${encodeURIComponent(q)}`, 8000);
   356→  }
   357→  if (!html) return [];
   358→  const out: SearchItem[] = [];
   359→  const re = /<a[^>]*class="[^"]*result__a[^"]*"[^>]*href="([^"]+)"[^>]*>([\s\S]*?)<\/a>/g;
   360→  const snips = [...html.matchAll(/<a[^>]*class="[^"]*result__snippet[^"]*"[^>]*>([\s\S]*?)<\/a>/g)]
   361→    .map(m2 => stripTags(m2[1]));
   362→  let m: RegExpExecArray | null;
   363→  while ((m = re.exec(html)) && out.length < 8) {
   364→    let url = m[1];
   365→    const uddg = url.match(/[?&]uddg=([^&]+)/);
   366→    if (uddg) { try { url = decodeURIComponent(uddg[1]); } catch { /* keep raw */ } }
   367→    const name = stripTags(m[2]);
   368→    if (!name || !/^https?:\/\//.test(url) || /duckduckgo\.com/.test(url)) continue;
   369→    out.push({ name, snippet: snips[out.length] || name, url });
   370→  }
   371→  return out;
   372→}

   374→// hosted search index — last resort when both scrapes come up empty
   375→async function hostedSearch(q: string): Promise<SearchItem[]> {
   376→  try {
   377→    const zai = await ZAI.create();
   378→    const res = await Promise.race([
   379→      zai.functions.invoke("web_search", { query: q.slice(0, 300), num: 8 }),
   380→      new Promise<unknown>(r => setTimeout(() => r(null), 12000)),
   381→    ]);
   382→    const items = Array.isArray(res) ? (res as { name?: unknown; snippet?: unknown; url?: unknown; date?: unknown }[]) : [];
   383→    return items
   384→      .map(it => ({
   385→        name: String(it.name || "").slice(0, 300),
   386→        snippet: String(it.snippet || "").slice(0, 500),
   387→        url: String(it.url || ""),
   388→        date: it.date ? String(it.date) : undefined,

   396→// merge: Google News first, DuckDuckGo web second, hosted index as the safety net
   397→async function googleSearch(q: string): Promise<SearchItem[]> {
   398→  const [news, web] = await Promise.all([googleNews(q), duckWeb(q)]);
   399→  const seen = new Set<string>();
   400→  const merged: SearchItem[] = [];
   401→  for (const it of [...news, ...web]) {
   402→    const key = it.url.replace(/[?#].*$/, "").toLowerCase();
   403→    if (!it.url || seen.has(key)) continue;
   404→    seen.add(key);
   405→    merged.push(it);
   406→    if (merged.length >= 8) break;
   407→  }
   408→  if (merged.length < 3) {
   409→    for (const it of await hostedSearch(q)) {
   410→      const key = it.url.replace(/[?#].*$/, "").toLowerCase();
   411→      if (seen.has(key)) continue;
   412→      seen.add(key);
   413→      merged.push(it);
   414→      if (merged.length >= 8) break;
   415→    }
   416→  }
   417→  return merged;
   418→}

   495→async function glmChat(sys: string, msgs: { role: "user" | "assistant"; content: string }[], model: string): Promise<ChatResult> {
   496→  // self-healing: up to 2 attempts, re-creating the engine client each time —
   497→  // a stale/poisoned SDK instance (env refresh after recycle) recovers instantly.
   498→  // 429 short-circuit: return quota:true at once — the CALLER decides whether
   499→  // to probe the other free lanes (FALLBACK_MODELS) or answer from Google /
   500→  // honest outage. No unbounded retries here: one message must never spin
   501→  // 16+ gateway calls.
   502→  for (let attempt = 0; attempt < 2; attempt++) {
   503→    try {
   504→      const zai = await ZAI.create();
   505→      // thinking toggle is a GLM knob — applied to glm-* models only, never
   506→      // to the emergency dpsk/argon lanes
   507→      const glmish = model.startsWith("glm");
   508→      const res = (await Promise.race([
   509→        zai.chat.completions.create({
   510→          model,
   511→          messages: [{ role: "system", content: sys }, ...msgs],
   512→          ...(glmish ? { thinking: { type: "disabled" as const } } : {}),
   513→        }),
   514→        new Promise<null>(r => setTimeout(() => r(null), 60000)),
   515→      ])) as { choices?: { message?: { content?: string } }[] } | null;
   516→      const text = String(res?.choices?.[0]?.message?.content || "").trim();
   517→      if (text) return { text, quota: false };
   518→      // empty reply is an engine hiccup, not a quota signal — retry once
   519→    } catch (e) {
   520→      const msg = String((e as Error)?.message || e);
   521→      if (/\b429\b|too many requests/i.test(msg)) return { text: null, quota: true };
   522→      // engine hiccup — fall through to retry
   523→    }
   524→    if (attempt < 1) await new Promise(r => setTimeout(r, 1200));
   525→  }
   526→  return { text: null, quota: false }; // caller tries the next engine
   527→}

   544→// pull plain text out of string-or-multipart content
   545→function textOf(content: unknown): string {
   546→  if (typeof content === "string") return content;
   547→  if (Array.isArray(content)) {
   548→    return content
   549→      .map(p => (p && typeof p === "object" && typeof (p as Part).text === "string" ? ((p as Part).text as string) : ""))
   550→      .join("\n");
   551→  }
   552→  return "";
   553→}

   555→// ---- real vision preprocessing ----
   556→// pull base64 image blocks (photos + video keyframes) out of the last user message
   557→function imagesOf(content: unknown): { mt: string; b64: string }[] {
   558→  if (!Array.isArray(content)) return [];
   559→  const out: { mt: string; b64: string }[] = [];
   560→  for (const p of content) {
   561→    const part = p as Part;
   562→    const src = part?.source;
   563→    if (part?.type === "image" && src && typeof src.data === "string" && src.data.length > 64) {
   564→      const mt =
   565→        src.media_type && /^image\/(png|jpe?g|webp|gif)$/.test(src.media_type)
   566→          ? (src.media_type === "image/jpg" ? "image/jpeg" : src.media_type)
   567→          : "image/jpeg";
   568→      if (src.data.length <= 4_200_000) out.push({ mt, b64: src.data });
   569→    }
   570→  }
   571→  return out.slice(0, 8);
   572→}

   821→      await new Promise(r2 => setTimeout(r2, 3000));
   822→      const retry = await glmChat(sys, chatText, chosen);
   823→      if (retry.text) {
   824→        const a2 = parseAction(retry.text);
   825→        if (a2) { actText = retry.text; engineUsed = chosen; act = a2; break; }
   826→        return NextResponse.json({ ok: true, text: withSources(retry.text, mode, lastSearch), engine: chosen, mode, vision: visionBlock || undefined });
   827→      }
   828→      continue;
   829→    }
   830→    if (!r.text) continue;
   831→    const t = r.text;
   832→    const a = parseAction(t);
   833→    if (a) {
   834→      actText = t;
   835→      engineUsed = engine;
   836→      act = a;
   837→      break;
   838→    }
   839→    return NextResponse.json({ ok: true, text: withSources(t, mode, lastSearch), engine, mode, vision: visionBlock || undefined });
   840→  }
   841→  if (!actText || !act) {
   842→    const out = googleOutage();
   843→    if (out) return out;
   844→    return NextResponse.json({ ok: true, text: outageReply(mode), engine: "xr-code", mode, vision: visionBlock || undefined });
   845→  }
   846→
   847→  // creation request — build the real asset, then present it in one natural line
   848→  const asset =
   849→    act.action === "image"
   850→      ? await runImageGen((act as { prompt: string }).prompt, (act as { size?: string }).size)
   851→      : await runVideoGen((act as { prompt: string }).prompt, (act as { with_audio?: boolean }).with_audio === true);
   852→  const composerMsgs = [
   853→    ...chatText,
   854→    { role: "assistant" as const, content: JSON.stringify(act) },
   855→    {
   856→      role: "user" as const,
   857→      content: asset
   858→        ? `Tool result: the ${act.action} was generated successfully and is attached in the chat. ` +
   859→          "Write ONE short friendly sentence presenting it to the user — never mention JSON, tools, URLs or engines."
   860→        : `Tool result: the ${act.action} generation failed or timed out this time. ` +
   861→          "Apologize in ONE short natural sentence and suggest trying again in a moment. Never mention JSON, tools or engines.",
   862→    },
   863→  ];
   864→  const composed = await glmChat(SYSTEM_KODER, composerMsgs, engineUsed);
   865→  const text = composed.text ||
   866→    (asset ? "Here it is — freshly generated for you." : "The generation didn't finish this time — give it another try in a moment.");
   867→  return NextResponse.json({
   868→    ok: true,
   869→    text,
   870→    engine: engineUsed,
   871→    mode,
   872→    asset: asset ? { type: act.action, ...asset, prompt: (act as { prompt: string }).prompt } : undefined,
   873→    vision: visionBlock || undefined,
   874→  });
