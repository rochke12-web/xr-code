   134→function parseAction(reply: string | null): Action | null {
   135→  if (!reply) return null;
   136→  const s = reply.trim().replace(/^```[a-z]*\s*/i, "").replace(/\s*```$/, "").trim();
   137→  if (s.length > 600 || !s.startsWith("{")) return null;
   138→  try {
   139→    const o = JSON.parse(s) as Record<string, unknown>;
   140→    if (
   141→      (o.action === "image" || o.action === "video") &&
   142→      typeof o.prompt === "string" && o.prompt.trim().length > 2
   143→    ) {
   144→      return {
   145→        action: o.action,
   146→        prompt: o.prompt.trim().slice(0, 800),
   147→        size: typeof o.size === "string" ? o.size : undefined,
   148→        with_audio: o.with_audio === true,
   149→      };
   150→    }
   151→  } catch {
   152→    /* normal text */
   153→  }
   154→  return null;
   155→}

   181→  try {
   182→    // create the task — up to 3 attempts; on 429 rate-limit wait 45s before the next one
   183→    let task: { id?: string } | null = null;
   184→    for (let a = 0; a < 3 && !task; a++) {

   197→        ])) as { id?: string } | null;

   199→      } catch (e) {

   239→// does this message deserve live web results?
   240→// knowledge / current-info questions — yes. pure code tasks and math/reasoning — no.
   241→function wantsWeb(q: string): boolean {
   242→  const s = q.toLowerCase();
   243→  if (s.length < 4) return false;
   244→  // identity / meta questions about the AI itself — answer directly, never search
   245→  if (/(who are you|what are you|your name|who made you|who created you|who built you|about yourself|who are u)/.test(s)) return false;
   246→  const questionWord = /^(who|what|when|where|why|how|which|whose|is|are|was|were|do|does|did|can|could|should|define|explain|tell me about)\b/.test(s);
   247→  const currentInfo = /(latest|current|today|right now|news|price of|net worth|population|how old|release date|who is|what is)/.test(s);
   248→  const followUp = /^(and|also|his|her|its|their|what about|how about)\b/.test(s) && s.includes("?");
   249→  if (!(questionWord || currentInfo || followUp)) return false;
   250→  // pure code tasks never search, even if phrased oddly long
   251→  if (/^(write|create|make|build|fix|refactor|optimi[sz]e|implement|add)\b/.test(s) && !s.includes("?") && !/who is|what is|latest|current|news/.test(s)) return false;
   252→  return true;
   253→}

   255→// context-aware search query: short follow-ups like "how old is he?" inherit
   256→// the previous question's topic so the search still finds the right thing.
   257→function searchQuery(safe: Msg[]): string {
   258→  const users = safe.filter(m => m.role === "user");
   259→  const last = (users[users.length - 1]?.content ?? "").trim();
   260→  if (
   261→    users.length >= 2 &&
   262→    (last.length < 45 || /^(and|also|his|her|its|their|what about|how about|then|so)\b/i.test(last))
   263→  ) {
   264→    const prev = users[users.length - 2].content.slice(0, 120);
   265→    return `${prev} ${last}`.slice(0, 300);
   266→  }
   267→  return last.slice(0, 300);
   268→}

   344→async function glmChat(sys: string, msgs: { role: "user" | "assistant"; content: string }[], model: string): Promise<string | null> {
   345→  // self-healing: up to 3 attempts, re-creating the engine client each time —
   346→  // a stale/poisoned SDK instance (env refresh after recycle) recovers instantly
   347→  for (let attempt = 0; attempt < 3; attempt++) {
   348→    try {
   349→      const zai = await ZAI.create();
   350→      const res = (await Promise.race([
   351→        zai.chat.completions.create({
   352→          model,
   353→          messages: [{ role: "system", content: sys }, ...msgs],
   354→          thinking: { type: "disabled" as const },
   355→        }),
   356→        new Promise<null>(r => setTimeout(() => r(null), 60000)),
   357→      ])) as { choices?: { message?: { content?: string } }[] } | null;
   358→      const text = String(res?.choices?.[0]?.message?.content || "").trim();
   359→      if (text) return text;
   360→    } catch {
   361→      // engine hiccup — fall through to retry
   362→    }
   363→    if (attempt < 2) await new Promise(r => setTimeout(r, 1200));
   364→  }
   365→  return null; // caller tries the next engine
   366→}

   379→// ---- real vision preprocessing ----

   381→function imagesOf(content: unknown): { mt: string; b64: string }[] {
   382→  if (!Array.isArray(content)) return [];
   383→  const out: { mt: string; b64: string }[] = [];
   384→  for (const p of content) {
   385→    const part = p as Part;
   386→    const src = part?.source;
   387→    if (part?.type === "image" && src && typeof src.data === "string" && src.data.length > 64) {
   388→      const mt =
   389→        src.media_type && /^image\/(png|jpe?g|webp|gif)$/.test(src.media_type)
   390→          ? (src.media_type === "image/jpg" ? "image/jpeg" : src.media_type)
   391→          : "image/jpeg";
   392→      if (src.data.length <= 4_200_000) out.push({ mt, b64: src.data });
   393→    }
   394→  }
   395→  return out.slice(0, 8);
   396→}

   412→    ];
   413→    for (const im of imgs) {
   414→      content.push({ type: "image_url", image_url: { url: `data:${im.mt};base64,${im.b64}` } });
   415→    }
   416→    // SDK types demand `model` and omit `role`, but the verified-working runtime
   417→    // payload needs neither — cast instead of changing what works.
   418→    const visionBody = {
   419→      messages: [{ role: "user", content }],
   420→      thinking: { type: "disabled" as const },
   421→    };
   422→    const res = (await Promise.race([
   423→      zai.chat.completions.createVision(
   424→        visionBody as unknown as Parameters<typeof zai.chat.completions.createVision>[0]
   425→      ),
   426→      new Promise<null>(r => setTimeout(() => r(null), 32000)),
   427→    ])) as { choices?: { message?: { content?: string } }[] } | null;
   428→    return String(res?.choices?.[0]?.message?.content || "").trim();
   429→  } catch {
   430→    return ""; // vision preprocessing down → fall back to blind text notes
   431→  }
   432→}

   438→export async function POST(req: Request) {
   439→  let body: { messages?: RawMsg[]; model?: string; mode?: string } | null = null;
   440→  try {
   441→    body = await req.json();
   442→  } catch {
   443→    /* no / invalid body */
   444→  }
   445→  const raw = Array.isArray(body?.messages) ? (body!.messages as RawMsg[]) : null;
   446→  const safe: Msg[] = raw
   447→    ? raw
   448→        .slice(-24)
   449→        .filter(m => m && (m.role === "user" || m.role === "assistant") && m.content != null)
   450→        .map(m => ({ role: m.role as "user" | "assistant", content: textOf(m.content).slice(0, 8000) }))
   451→    : [];
   452→  if (!safe.length) {
   453→    return NextResponse.json({ ok: false, error: "bad request" }, { status: 400 });
   454→  }
   455→
   456→  // two modes: Koder (developer brain) and QnA (web-grounded answers, sources stay hidden)
   457→  const modeRaw = typeof body?.mode === "string" ? body.mode.trim().toLowerCase() : "";
   458→  const mode = modeRaw === "qna" || modeRaw === "sight" ? modeRaw : "koder";
   459→  let sys = (mode === "qna" ? SYSTEM_QNA : mode === "sight" ? SYSTEM_SIGHT : SYSTEM_KODER) + TOOL_RULES;
   460→  const lastUser = [...safe].reverse().find(m => m.role === "user");
   461→  // one brain, all GLM: explicit family member is honored, otherwise auto-route
   462→  const wanted = typeof body?.model === "string" ? body.model.trim() : "";
   463→  const chosen = GLM_MODELS.has(wanted) ? wanted : autoEngine(lastUser?.content || "", safe);
   464→  // the user is talking about THEIR attachment — web search would only distract.
   465→  // QnA mode grounds almost every real question in live results (sources never shown).
   466→  if (lastUser && !/\[attached (image|video|file|audio):/i.test(lastUser.content)) {
   467→    const q = lastUser.content.toLowerCase();
   468→    const identityQ = /(who are you|what are you|your name|who made you|who created you|who built you|about yourself|who are u)/.test(q);
   469→    const looksQuestion = q.length >= 4 && (/\?/.test(q) || /^(who|what|when|where|why|how|which|whose|is|are|was|were|do|does|did|can|could|should|define|explain|tell me)\b/.test(q) || q.length > 60);
   470→    const forceWeb = mode === "qna" && !identityQ && looksQuestion;
   471→    if (forceWeb || wantsWeb(lastUser.content)) {
   472→      sys += await googleContext(searchQuery(safe));
   473→      if (mode === "qna") {
   474→        sys += "\n\nSECRECY REMINDER: answer from the LIVE WEB RESULTS as your own knowledge — never mention sources, " +
   475→               "sites, links or that anything was searched. No Sources line, even if the user asks for one.";
   476→      } else if (/\b(sources?|cite|citations?|references?|links?|urls?)\b/i.test(lastUser.content)) {
   477→        sys += "\n\nThe user explicitly asked for sources: end your reply with a final line 'Sources:' " +
   478→               "followed by up to 3 of the LIVE WEB RESULTS URLs above.";
   479→      }
   480→    }
   481→  }
   482→
   483→  // ---- real vision: describe the attached photos / video keyframes, then feed
   484→  // the description to the text brain as a [VISION] block ----
   485→  let visionBlock = "";
   486→  const rawLastUser = raw ? [...raw].reverse().find(m => m && m.role === "user") : null;
   487→  const imgs = imagesOf(rawLastUser?.content);
   488→  if (imgs.length) {
   489→    const desc = await describeImages(imgs);
   490→    if (desc) {
   491→      visionBlock = "\n\n[VISION — what the attached images show]\n" + desc + "\n[end VISION]";
   492→      sys = sys.replace(ATT_BLIND, ATT_VISION);
   493→    }
   494→  }
   495→
   496→  const chatText = safe.map(m => ({ role: m.role, content: m.content }));
   497→  if (visionBlock) {
   498→    const lu = [...chatText].reverse().find(m => m.role === "user");
   499→    if (lu) lu.content = (lu.content + visionBlock).slice(0, 12000);
   500→  }
   501→
   502→  // one GLM brain — plain replies return; action directives run the tools
   503→  const chain = [chosen, ...GLM_FALLBACKS.filter(m => m !== chosen)];
   504→  let actText: string | null = null;
   505→  let engineUsed = "";
   506→  let act: Action | null = null;
   507→  for (const engine of chain) {
   508→    const t = await glmChat(sys, chatText, engine);
   509→    if (!t) continue;
   510→    const a = parseAction(t);
   511→    if (a) {
   512→      actText = t;
   513→      engineUsed = engine;
   514→      act = a;
   515→      break;
   516→    }
   517→    return NextResponse.json({ ok: true, text: t, engine, mode, vision: visionBlock || undefined });
   518→  }
   519→  if (!actText || !act)
   520→    return NextResponse.json({ ok: false, error: "engine unreachable" });
   521→
   522→  // creation request — build the real asset, then present it in one natural line
   523→  const asset =
   524→    act.action === "image"
   525→      ? await runImageGen(act.prompt, act.size)
   526→      : await runVideoGen(act.prompt, act.with_audio === true);
   527→  const composerMsgs = [
   528→    ...chatText,
   529→    { role: "assistant" as const, content: JSON.stringify(act) },
   530→    {
   531→      role: "user" as const,
   532→      content: asset
   533→        ? `Tool result: the ${act.action} was generated successfully and is attached in the chat. ` +
   534→          "Write ONE short friendly sentence presenting it to the user — never mention JSON, tools, URLs or engines."
   535→        : `Tool result: the ${act.action} generation failed or timed out this time. ` +
   536→          "Apologize in ONE short natural sentence and suggest trying again in a moment. Never mention JSON, tools or engines.",
   537→    },
   538→  ];
   539→  const text = (await glmChat(SYSTEM_KODER, composerMsgs, engineUsed)) ||
   540→    (asset ? "Here it is — freshly generated for you." : "The generation engine didn't finish this time — give it another try in a moment.");
   541→  return NextResponse.json({
   542→    ok: true,
   543→    text,
   544→    engine: engineUsed,
   545→    mode,
   546→    asset: asset ? { type: act.action, ...asset, prompt: act.prompt } : undefined,
   547→    vision: visionBlock || undefined,
   548→  });
   549→}
