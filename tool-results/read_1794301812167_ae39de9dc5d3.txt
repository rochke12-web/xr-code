   162→   162→  if (s.length > 600 || !s.startsWith("{")) return null;
   163→   163→  try {
   164→   164→    const o = JSON.parse(s) as Record<string, unknown>;
   165→   165→    if (o.action === "search" && typeof o.query === "string" && o.query.trim().length > 1) {
   166→   166→      return { action: "search", query: o.query.trim().slice(0, 200) };
   167→   167→    }
   168→   168→    if (
   169→   169→      (o.action === "image" || o.action === "video") &&
   170→   170→      typeof o.prompt === "string" && o.prompt.trim().length > 2

   178→   178→    }

   267→   267→// does this message deserve live web results?

   269→   269→function wantsWeb(q: string): boolean {
   270→   270→  const s = q.toLowerCase();
   271→   271→  if (s.length < 4) return false;
   272→   272→  // identity / meta questions about the AI itself — answer directly, never search
   273→   273→  if (/(who are you|what are you|your name|who made you|who created you|who built you|about yourself|who are u)/.test(s)) return false;
   274→   274→  const questionWord = /^(who|what|when|where|why|how|which|whose|is|are|was|were|do|does|did|can|could|should|define|explain|tell me about)\b/.test(s);
   275→   275→  const currentInfo = /(latest|current|today|right now|news|price of|net worth|population|how old|release date|who is|what is)/.test(s);
   276→   276→  const followUp = /^(and|also|his|her|its|their|what about|how about)\b/.test(s) && s.includes("?");
   277→   277→  if (!(questionWord || currentInfo || followUp)) return false;
   278→   278→  // pure code tasks never search, even if phrased oddly long
   279→   279→  if (/^(write|create|make|build|fix|refactor|optimi[sz]e|implement|add)\b/.test(s) && !s.includes("?") && !/who is|what is|latest|current|news/.test(s)) return false;
   280→   280→  return true;
   281→   281→}

   331→   331→    return null;

   338→   338→    `https://news.google.com/rss/search?q=${encodeURIComponent(q)}&hl=en-US&gl=US&ceid=US:en`, 9000);
   339→   339→  if (!xml) return [];
   340→   340→  const out: SearchItem[] = [];
   341→   341→  for (const chunk of xml.split("<item>").slice(1, 9)) {
   342→   342→    const name = stripTags(chunk.match(/<title>([\s\S]*?)<\/title>/)?.[1] || "");
   343→   343→    const url = (chunk.match(/<link>([\s\S]*?)<\/link>/)?.[1] || "").trim();
   344→   344→    const date = (chunk.match(/<pubDate>([\s\S]*?)<\/pubDate>/)?.[1] || "").trim();
   345→   345→    if (!name || !url) continue;
   346→   346→    out.push({ name, snippet: name, url, date: date ? date.slice(5, 16) : undefined });
   347→   347→  }
   348→   348→  return out;
   349→   349→}

   351→   351→// DuckDuckGo HTML — general-web gap filler (Google News skews to current events)
   352→   352→async function duckWeb(q: string): Promise<SearchItem[]> {
   353→   353→  let html: string | null = null;
   354→   354→  for (let i = 0; i < 2 && !html; i++) {
   355→   355→    html = await fetchPage(`https://html.duckduckgo.com/html/?q=${encodeURIComponent(q)}`, 8000);
   356→   356→  }
   357→   357→  if (!html) return [];
   358→   358→  const out: SearchItem[] = [];
   359→   359→  const re = /<a[^>]*class="[^"]*result__a[^"]*"[^>]*href="([^"]+)"[^>]*>([\s\S]*?)<\/a>/g;
   360→   360→  const snips = [...html.matchAll(/<a[^>]*class="[^"]*result__snippet[^"]*"[^>]*>([\s\S]*?)<\/a>/g)]
   361→   361→    .map(m2 => stripTags(m2[1]));
   362→   362→  let m: RegExpExecArray | null;
   363→   363→  while ((m = re.exec(html)) && out.length < 8) {
   364→   364→    let url = m[1];
   365→   365→    const uddg = url.match(/[?&]uddg=([^&]+)/);
   366→   366→    if (uddg) { try { url = decodeURIComponent(uddg[1]); } catch { /* keep raw */ } }
   367→   367→    const name = stripTags(m[2]);
   368→   368→    if (!name || !/^https?:\/\//.test(url) || /duckduckgo\.com/.test(url)) continue;
   369→   369→    out.push({ name, snippet: snips[out.length] || name, url });
   370→   370→  }
   371→   371→  return out;
   372→   372→}

   397→   397→async function googleSearch(q: string): Promise<SearchItem[]> {
   398→   398→  const [news, web] = await Promise.all([googleNews(q), duckWeb(q)]);

   408→   408→  if (merged.length < 3) {
   409→   409→    for (const it of await hostedSearch(q)) {
   410→   410→      const key = it.url.replace(/[?#].*$/, "").toLowerCase();
   411→   411→      if (seen.has(key)) continue;
   412→   412→      seen.add(key);
   413→   413→      merged.push(it);
   414→   414→      if (merged.length >= 8) break;
   415→   415→    }
   416→   416→  }
   417→   417→  return merged;
   418→   418→}

   473→   473→  const s = (q || "").toLowerCase();

   488→   488→async function glmChat(sys: string, msgs: { role: "user" | "assistant"; content: string }[], model: string): Promise<ChatResult> {
   489→   489→  // self-healing: up to 2 attempts, re-creating the engine client each time —
   490→   490→  // a stale/poisoned SDK instance (env refresh after recycle) recovers instantly.
   491→   491→  // 429 short-circuit: the whole family shares one key quota — once the gateway
   492→   492→  // says "too many requests" for any member, every other member will too, and
   493→   493→  // walking the rest of the chain only deepens the storm (16+ calls per message).
   494→   494→  // Detect it and bail out at once; the caller decides what to tell the user.
   495→   495→  for (let attempt = 0; attempt < 2; attempt++) {
   496→   496→    try {
   497→   497→      const zai = await ZAI.create();
   498→   498→      // thinking toggle is a GLM-specific knob — don't send it to other
   499→   499→      // families (dpsk/argon may reject unknown params)
   500→   500→      const glmish = model.startsWith("glm");
   501→   501→      const res = (await Promise.race([
   502→   502→        zai.chat.completions.create({
   503→   503→          model,
   504→   504→          messages: [{ role: "system", content: sys }, ...msgs],
   505→   505→          ...(glmish ? { thinking: { type: "disabled" as const } } : {}),
   506→   506→        }),
   507→   507→        new Promise<null>(r => setTimeout(() => r(null), 60000)),
   508→   508→      ])) as { choices?: { message?: { content?: string } }[] } | null;
   509→   509→      const text = String(res?.choices?.[0]?.message?.content || "").trim();
   510→   510→      if (text) return { text, quota: false };
   511→   511→      // empty reply is an engine hiccup, not a quota signal — retry once
   512→   512→    } catch (e) {
   513→   513→      const msg = String((e as Error)?.message || e);
   514→   514→      if (/\b429\b|too many requests/i.test(msg)) return { text: null, quota: true };
   515→   515→      // engine hiccup — fall through to retry
   516→   516→    }
   517→   517→    if (attempt < 1) await new Promise(r => setTimeout(r, 1200));
   518→   518→  }
   519→   519→  return { text: null, quota: false }; // caller tries the next engine
   520→   520→}
